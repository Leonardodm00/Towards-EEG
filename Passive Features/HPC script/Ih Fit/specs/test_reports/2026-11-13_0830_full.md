import numpy as np, noise_windows as nw
fs = 50000.0
def ss_windows(sweeps=(31,), K=6, flat_pre=False, seed=0):
    rng = np.random.default_rng(seed); n_w, pre = 10500, 500
    pos = np.arange(K); sw = np.repeat(np.asarray(sweeps), K // len(sweeps))
    v = -70.0 + 0.05 * rng.standard_normal((K, n_w))
    if flat_pre: v[:, :pre] = -70.0
    return nw.SSWindows(specimen_id=1, fs=fs, dt=1 / fs, t=(np.arange(n_w) - pre) / fs, v=v,
                        polarity_is_dep=(pos % 2 == 0), sweep_number=sw, position=pos % 37, train_length=37,
                        period_samples=10025, pre_n=pre, n_raw=37 * len(sweeps), n_sweeps=len(sweeps))
def ls_windows(sweeps=(60,), fs_ls=fs, n_pre=13250, seed=1):
    rng = np.random.default_rng(seed); k = len(sweeps)
    return nw.LSWindows(specimen_id=1, fs=fs_ls, dt=1 / fs_ls, sweep_number=np.asarray(sweeps), amplitude_pA=np.full(k, -50.0),
                        polarity=np.array(["hyp"] * k), pre=[-70 + 0.05 * rng.standard_normal(n_pre) for _ in range(k)],
                        n_pre=np.full(k, n_pre), onset_s=np.full(k, 0.27), end=[np.zeros(3)] * k,

                        end_status=np.array(["ok"] * k))
def tilts(protocol, sweeps):
